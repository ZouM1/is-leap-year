闰年判断器

from pyscript import when, document def is_leap_year(year): if (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0): return True else: return False @when("click", "#checkBtn") def check_leap(event): year_input = document.getElementById("yearInput").value result_element = document.getElementById("result") if not year_input: result_element.textContent = "请输入一个年份！" result_element.style.color = "#ff9800" return try: year = int(year_input) if is_leap_year(year): result_element.textContent = f"✅ {year}年 是闰年" result_element.style.color = "#4CAF50" else: result_element.textContent = f"❌ {year}年 不是闰年" result_element.style.color = "#f44336" except ValueError: result_element.textContent = "❌ 请输入有效的整数年份！" result_element.style.color = "#f44336"
